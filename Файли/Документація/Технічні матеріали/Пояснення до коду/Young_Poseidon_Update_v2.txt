  1. Керування повністю переназначено (W=стрибок, S=проскочити вниз,
     E=кидок, F=взаємодія, Z/X/C=зілля, 1-6=зброя)
  2. Зілля здоров'я відновлює 5 HP (було 2), Геракл дає +5 ATK
  3. Боси повністю переписані — у кожного унікальна механіка
  4. Порядок нагород: Тритон→Тризубець, Німфа→Спис,
     Уран→Кайло, Харон→Коса, Мінотавр→Весло,
     Привид→Меч, Мойри→Ключ
  5. Додано Зони коментарів, S-drop через платформи

ЗМІСТ:
  1.  obj_player        — Create, Step (повна версія)
  2.  obj_comment_zone  — Зона коментарів (нова!)

  5.  obj_boss_uranus   — Повна переробка (завали+кайло)
  6.  obj_boss_charon   — Повна переробка (човен+коса-бумеранг)

  10. Нагороди зброї    — Оновлений порядок

  1. OBJ_PLAYER — Create Event (ОНОВЛЕНО)
═══════════════════════════════════════════════════════════════════

─────────────────────────────────────────
/// @description obj_player — Create Event
─────────────────────────────────────────
// ════ ЗДОРОВ'Я ════
hp     = 5;
max_hp = 5;

// ════ ЗІЛЛЯ (клавіші Z / X / C) ════
potion_heal_count     = 0;   // Z — зілля здоров'я (+5 HP)
potion_strength_count = 0;   // X — зілля Геракла  (+5 ATK на 60с)
potion_speed_count    = 0;   // C — зілля Гермеса  (+1 SPD на 60с)

// ════ ХАРАКТЕРИСТИКИ ════
spd            = 1;       // Базова швидкість (sp=1 з документу)
perm_spd_bonus = 0;       // +1 за кожен сандаль Гермеса (макс 2)
temp_spd_buff  = 0;       // +1 від зілля Гермеса
temp_atk_buff  = 0;       // +5 від зілля Геракла
buff_duration  = 0;       // Кадри до кінця тимчасового бафу

// ════ ФІЗИКА ════
hspd = 0;    // Горизонтальна швидкість (knockback)
vspd = 0;    // Вертикальна швидкість
grv  = 0.5;  // Гравітація

// ════ СТАН ════
l_right      = true;
is_damage    = false;
is_frozen    = false;   // Заморозка від Тіні Аїда
freeze_timer = 0;

// ════ ЗБРОЯ ════
// Індекси: 0=Тризубець 1=Спис 2=Кайло 3=Коса 4=Весло 5=Меч
weapon_names    = ["Тризубець","Спис","Кайло","Коса","Весло","Меч"];
weapon_sprites  = [spr_weapon_trident, spr_weapon_spear,
                   spr_weapon_pickaxe, spr_weapon_scythe,
                   spr_weapon_oar,     spr_weapon_sword];

// ATK і множник SPD — точно з дизайн-документу
weapon_atk      = [1, 2, 5, 3, 4, 10];
weapon_spd_mult = [1.0, 2.0, 0.25, 1.0, 0.5, 2.0];

// Розблокована зброя (false = ще не отримана)
weapon_unlocked = [false, false, false, false, false, false];
// На початку гравець БЕЗ ЗБРОЇ — отримає Тризубець від Тритона

current_weapon = -1;  // -1 = немає зброї

// ════ ТРИЗУБЕЦЬ / СПИС ════
trident_instance = noone;

// ════ КЛЮЧ ОЛІМПУ ════
has_olympus_key = false;

// ════ ФУНКЦІЇ ════
_unlock_weapon = function(_index)
{
    weapon_unlocked[_index] = true;
    current_weapon          = _index;
};

_switch_weapon = function(_delta)
{
    if (current_weapon == -1) exit;
    var _n = array_length(weapon_unlocked);
    var _next = current_weapon;
    repeat (_n)
    {
        _next = (_next + _delta + _n) mod _n;
        if (weapon_unlocked[_next]) { current_weapon = _next; break; }
    }
};


═══════════════════════════════════════════════════════════════════
  1b. OBJ_PLAYER — Step Event (ОНОВЛЕНО)
      Нові клавіші з дизайн-документу:
      W/↑ = стрибок,  S/↓ = проскочити крізь платформу,
      E   = кидок/кинути зброю,  F = активувати предмет,
      1-6 = вибір зброї,  Z = здоров'я, X = Геракл, C = Гермес
═══════════════════════════════════════════════════════════════════

─────────────────────────────────────────
/// @description obj_player — Step Event
─────────────────────────────────────────
// ════════════════════════════════════════
// БЛОК 1: ЧИТАННЯ КЛАВІШ
// ════════════════════════════════════════
var _left  = keyboard_check(ord("A")) || keyboard_check(vk_left);
var _right = keyboard_check(ord("D")) || keyboard_check(vk_right);
var _jump  = keyboard_check_pressed(ord("W")) || keyboard_check_pressed(vk_up);
var _drop  = keyboard_check_pressed(ord("S")) || keyboard_check_pressed(vk_down);

var _e_pressed = keyboard_check_pressed(ord("E"));   // Кидок зброї
var _f_pressed = keyboard_check_pressed(ord("F"));   // Взаємодія з предметом
var _q_pressed = keyboard_check_pressed(ord("Q"));   // Закрити діалог

// Зілля
var _z_pressed = keyboard_check_pressed(ord("Z"));   // Зілля здоров'я
var _x_pressed = keyboard_check_pressed(ord("X"));   // Зілля Геракла
var _c_pressed = keyboard_check_pressed(ord("C"));   // Зілля Гермеса

// Колесо миші — переключення зброї
var _wheel_up   = mouse_wheel_up();
var _wheel_down = mouse_wheel_down();

// Клавіші 1-6 — пряме переключення зброї
for (var _ki = 0; _ki < 6; _ki++)
{
    if (keyboard_check_pressed(ord(string(_ki + 1))))
    {
        if (weapon_unlocked[_ki])
            current_weapon = _ki;
    }
}

// Колесо миші
if (_wheel_up)   _switch_weapon(1);
if (_wheel_down) _switch_weapon(-1);

// ════════════════════════════════════════
// БЛОК 2: ЗАМОРОЗКА (Тінь Аїда)
// ════════════════════════════════════════
if (is_frozen)
{
    freeze_timer--;
    if (freeze_timer <= 0) { is_frozen = false; }
    _left = false; _right = false;
}

// ════════════════════════════════════════
// БЛОК 3: РУХ
// ════════════════════════════════════════
var _move = _right - _left;

var _weapon_spd  = (current_weapon >= 0) ? weapon_spd_mult[current_weapon] : 1.0;
var _current_spd = spd * _weapon_spd * 10 + (perm_spd_bonus + temp_spd_buff) * 10;
var _move_hspd   = _move * _current_spd;

if (_move > 0) { l_right = true;  image_xscale =  1; }
if (_move < 0) { l_right = false; image_xscale = -1; }

// ════════════════════════════════════════
// БЛОК 4: ГРАВІТАЦІЯ
// ════════════════════════════════════════
vspd += grv;
if (vspd > 20) vspd = 20;

// ════════════════════════════════════════
// БЛОК 5: ЗІТКНЕННЯ З ПЛАТФОРМАМИ
// ════════════════════════════════════════
var _on_ground = place_meeting(x, y + 1, obj_platform);

// --- S: Проскочити крізь тонку платформу ---
// (лише якщо платформа є obj_thin_platform — нащадок obj_platform)
if (_drop && _on_ground && place_meeting(x, y + 1, obj_thin_platform))
{
    y    += 8;          // Штовхаємо гравця вниз крізь платформу
    vspd  = 1;
}

// Горизонталь
var _total_hspd = _move_hspd + hspd;
if (place_meeting(x + _total_hspd, y, obj_platform))
{
    while (!place_meeting(x + sign(_total_hspd), y, obj_platform))
        x += sign(_total_hspd);
    _total_hspd = 0;
    hspd        = 0;
}
x += _total_hspd;

hspd *= 0.75;
if (abs(hspd) < 0.5) hspd = 0;

// Стрибок
if (_on_ground && _jump)
{
    vspd = -12;
    if (audio_exists(snd_jump)) audio_play_sound(snd_jump, 1, false);
}

// Вертикаль
if (place_meeting(x, y + vspd, obj_platform))
{
    while (!place_meeting(x, y + sign(vspd), obj_platform))
        y += sign(vspd);
    vspd = 0;
}
y += vspd;

// ════════════════════════════════════════
// БЛОК 6: ЗБРОЯ — E (КИДОК) і F (ПОВЕРНЕННЯ/ВЗАЄМОДія)
// ════════════════════════════════════════
var _can_throw = (current_weapon == 0 || current_weapon == 1);

if (_can_throw && instance_exists(trident_instance))
{
    var _t = trident_instance;

    // E — кинути зброю
    if (_e_pressed)
    {
        if (_t.stuck)
        {
            // Знімаємо зі стіни і кидаємо знову
            _t.stuck = false;
            var _n   = instance_number(obj_trident_platform);
            for (var _i = _n - 1; _i >= 0; _i--)
            {
                var _pl = instance_find(obj_trident_platform, _i);
                if (instance_exists(_pl)) instance_destroy(_pl);
            }
        }
        if (!_t.thrown && !_t.stuck)
        {
            _t.thrown       = true;
            _t.timer        = 0;
            _t.x            = x + image_xscale * 24;
            _t.y            = y - 8;
            _t.image_xscale = image_xscale;
        }
    }

    // F — повернути з будь-якої відстані (натискання "1" у документі)
    // У документі написано "цифру 1", але F — кнопка взаємодії
    // Залишаємо обидва варіанти:
    if (_f_pressed || keyboard_check_pressed(ord("1")))
    {
        if (_t.stuck || _t.thrown)
        {
            var _n = instance_number(obj_trident_platform);
            for (var _i = _n - 1; _i >= 0; _i--)
            {
                var _pl = instance_find(obj_trident_platform, _i);
                if (instance_exists(_pl)) instance_destroy(_pl);
            }
            _t.stuck       = false;
            _t.thrown      = false;
            _t.image_angle = 0;
        }
    }
}

// Меч — випад (E при current_weapon == 5)
if (current_weapon == 5 && _e_pressed)
{
    hspd = image_xscale * 18;
    var _hit = instance_place(x + image_xscale * 40, y, obj_enemy);
    if (instance_exists(_hit))
        with (_hit) hp -= (obj_player.weapon_atk[5] + obj_player.temp_atk_buff);
}

// ════════════════════════════════════════
// БЛОК 7: СМЕРТЬ
// ════════════════════════════════════════
if (hp <= 0)
{
    hp = max_hp;
    if (variable_global_exists("checkpoint_room") && global.checkpoint_room == room)
    {
        x = global.checkpoint_x;
        y = global.checkpoint_y;
        vspd = 0; hspd = 0;
    }
    else if (variable_global_exists("checkpoint_room"))
        room_goto(global.checkpoint_room);
    else
        game_restart();
}

// ════════════════════════════════════════
// БЛОК 8: ТИМЧАСОВІ БАФИ
// ════════════════════════════════════════
if (buff_duration > 0)
{
    buff_duration--;
    if (buff_duration <= 0) { temp_spd_buff = 0; temp_atk_buff = 0; }
}

// ════════════════════════════════════════
// БЛОК 9: ЗІЛЛЯ (Z / X / C)
// ════════════════════════════════════════

// Z — Зілля здоров'я: відновлює 5 HP (з дизайн-документу)
if (_z_pressed && potion_heal_count > 0 && hp < max_hp)
{
    hp = min(hp + 5, max_hp);
    potion_heal_count--;
    // audio_play_sound(snd_potion_heal, 1, false);
}

// X — Зілля Геракла: +5 ATK на 60 секунд (з дизайн-документу)
if (_x_pressed && potion_strength_count > 0 && buff_duration <= 0)
{
    potion_strength_count--;
    temp_atk_buff = 5;
    buff_duration = room_speed * 60;   // 60 секунд
    // audio_play_sound(snd_potion_strength, 1, false);
}

// C — Зілля Гермеса: +1 SPD на 60 секунд (з дизайн-документу)
if (_c_pressed && potion_speed_count > 0 && buff_duration <= 0)
{
    potion_speed_count--;
    temp_spd_buff = 1;
    buff_duration = room_speed * 60;   // 60 секунд
    // audio_play_sound(snd_potion_speed, 1, false);
}


═══════════════════════════════════════════════════════════════════
  2. OBJ_COMMENT_ZONE — Зона коментарів (НОВИЙ ОБ'ЄКТ)
     Розміщується в кімнаті. Коли гравець входить в зону —
     Посейдон щось коментує. Закривається клавішею Q.
═══════════════════════════════════════════════════════════════════

─────────────────────────────────────────
/// @description obj_comment_zone — Create Event
─────────────────────────────────────────
// ---- РОЗМІР ЗОНИ (встановлюється в кімнаті) ----
zone_width  = 120;    // Ширина зони спрацювання
zone_height = 200;    // Висота зони

// ---- ТЕКСТ ----
// Встановлюється через Instance Creation Code в кімнаті:
// comment_text = "Хм, тут темно...";
comment_text = "...";

// Підтримка кількох рядків — розбиваємо по "|"
// Приклад: "Перший рядок|Другий рядок|Третій рядок"
comment_lines = string_split(comment_text, "|");
current_line  = 0;    // Поточний рядок діалогу

// ---- СТАН ----
is_showing    = false;   // Зараз показується?
has_triggered = false;   // Вже спрацьовувало? (один раз)

// ---- АНІМАЦІЯ ----
box_alpha     = 0;       // Прозорість вікна (плавна поява)
box_alpha_target = 0;


─────────────────────────────────────────
/// @description obj_comment_zone — Step Event
─────────────────────────────────────────
if (!instance_exists(obj_player)) exit;

// ---- ПЕРЕВІРКА ВХОДУ В ЗОНУ ----
var _px = obj_player.x;
var _py = obj_player.y;
var _in_zone = (_px > x && _px < x + zone_width
             && _py > y && _py < y + zone_height);

// Запускаємо один раз коли гравець входить
if (_in_zone && !has_triggered)
{
    has_triggered = true;
    is_showing    = true;
    current_line  = 0;
    box_alpha_target = 1;
    // audio_play_sound(snd_comment, 1, false);
}

// ---- Q: НАСТУПНИЙ РЯДОК / ЗАКРИТИ ----
if (is_showing && keyboard_check_pressed(ord("Q")))
{
    current_line++;
    if (current_line >= array_length(comment_lines))
    {
        // Всі рядки показано — закриваємо
        is_showing       = false;
        box_alpha_target = 0;
    }
}

// Плавна зміна прозорості
box_alpha = lerp(box_alpha, box_alpha_target, 0.15);


─────────────────────────────────────────
/// @description obj_comment_zone — Draw GUI Event
─────────────────────────────────────────
if (!is_showing || box_alpha < 0.05) exit;

var _gw   = display_get_gui_width();
var _gh   = display_get_gui_height();
var _bw   = _gw * 0.7;
var _bh   = 90;
var _bx   = _gw / 2 - _bw / 2;
var _by   = _gh - _bh - 70;

// Тло вікна
draw_set_alpha(box_alpha * 0.88);
draw_set_color(make_color_rgb(10, 20, 40));
draw_rectangle(_bx, _by, _bx + _bw, _by + _bh, false);

// Контур
draw_set_alpha(box_alpha);
draw_set_color(make_color_rgb(80, 140, 220));
draw_rectangle(_bx, _by, _bx + _bw, _by + _bh, true);

// Ім'я персонажа
draw_set_halign(fa_left);
draw_set_valign(fa_top);
draw_set_color(make_color_rgb(100, 200, 255));
if (font_exists(fnt_main)) draw_set_font(fnt_main);
draw_text(_bx + 14, _by + 10, "Посейдон:");

// Текст коментаря
draw_set_color(c_white);
var _line_text = "";
if (current_line < array_length(comment_lines))
    _line_text = comment_lines[current_line];
draw_text_ext(_bx + 14, _by + 30, _line_text, 24, _bw - 28);

// Підказка "Q — далі"
var _has_more = (current_line < array_length(comment_lines) - 1);
draw_set_halign(fa_right);
draw_set_color(make_color_rgb(140, 160, 180));
draw_text(_bx + _bw - 10, _by + _bh - 20,
    _has_more ? "[Q] Далі ▶" : "[Q] Закрити ✕");

draw_set_alpha(1);
draw_set_halign(fa_left);
draw_set_color(c_white);
draw_set_font(-1);


═══════════════════════════════════════════════════════════════════
  3. OBJ_BOSS_TRITON — ПОВНА ПЕРЕРОБКА
     hp=50  atk=1  sp=2
     Гравець НЕ МАЄ ЗБРОЇ. Перемагає натискаючи кнопки
     щоб впало каміння на Тритона (10 шкоди за камінь).
     Тритон літає між 4 точками.
═══════════════════════════════════════════════════════════════════

─────────────────────────────────────────
/// @description obj_boss_triton — Create Event
─────────────────────────────────────────
boss_name = "ТРИТОН";
max_hp    = 50;
hp        = max_hp;
atk       = 1;     // З дизайн-документу
phase     = 1;

// ---- 4 ТОЧКИ ПАТРУЛЯ ----
var _cx = x; var _cy = y;
point_count = 4;
point_x[0] = _cx - 280;  point_y[0] = _cy - 120;
point_x[1] = _cx + 280;  point_y[1] = _cy - 120;
point_x[2] = _cx + 280;  point_y[2] = _cy + 80;
point_x[3] = _cx - 280;  point_y[3] = _cy + 80;
current_point = 0;
fly_speed     = 3;

// ---- КОНТАКТНА АТАКА ----
// Тритон завдає мінімальну шкоду при дотику (atk=1)
contact_cd     = 0;
contact_cd_max = 90;

is_dead = false;
image_speed = 0.1;

// ---- ПІДКАЗКА ДЛЯ ГРАВЦЯ ----
// obj_interactive_button вже розміщені в кімнаті.
// Коли гравець натискає кнопку — з'являється obj_falling_rocks
// який при попаданні в Тритона завдає 10 шкоди.
// Це реалізовано через User Event 0 на Тритоні нижче.


─────────────────────────────────────────
/// @description obj_boss_triton — Step Event
─────────────────────────────────────────
if (is_dead || !instance_exists(obj_player)) exit;

if (hp <= 0)
{
    is_dead = true;
    instance_create_layer(x, y, "Instances", obj_pickup_trident_reward);
    instance_destroy();
    exit;
}

if (contact_cd > 0) contact_cd--;

// ---- РУХ МІЖ ТОЧКАМИ ----
var _tx = point_x[current_point];
var _ty = point_y[current_point];
var _d  = point_distance(x, y, _tx, _ty);

if (_d > fly_speed)
{
    var _dir = point_direction(x, y, _tx, _ty);
    x += lengthdir_x(fly_speed, _dir);
    y += lengthdir_y(fly_speed, _dir);
    image_xscale = (lengthdir_x(1, _dir) > 0) ? 1 : -1;
}
else
{
    x = _tx; y = _ty;
    current_point = (current_point + 1) mod point_count;
}

// ---- КОНТАКТНА ШКОДА ----
if (contact_cd <= 0 && distance_to_object(obj_player) < 40
    && !obj_player.is_damage)
{
    obj_player.hp       -= atk;
    var _kd = point_direction(x, y, obj_player.x, obj_player.y);
    obj_player.hspd     = lengthdir_x(8, _kd);
    obj_player.vspd     = -5;
    obj_player.is_damage = true;
    obj_player.alarm[0]  = room_speed;
    contact_cd = contact_cd_max;
}


─────────────────────────────────────────
/// @description obj_boss_triton — User Event 0
// Викликається obj_falling_rocks коли попадає в Тритона.
// 10 шкоди за кожен камінь.
─────────────────────────────────────────
hp -= 10;
// Ефект попадання каміння
