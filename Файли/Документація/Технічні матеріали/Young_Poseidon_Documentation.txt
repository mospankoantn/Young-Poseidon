  2. СИСТЕМА ГРАВЦЯ
════════════════════════════════════════════════════════════════════════════════

  ОБ'ЄКТ: obj_player  |  Persistent: TRUE

  ── ХАРАКТЕРИСТИКИ ──────────────────────────────────────────────
  Змінна          Тип      Опис                        Початок
  ─────────────────────────────────────────────────────────────────
  hp              real     Поточне здоров'я            5
  max_hp          real     Максимальне здоров'я        5
  spd             real     Базова швидкість            1
  perm_spd_bonus  real     +SPD від сандалів (макс 2)  0
  temp_spd_buff   real     +SPD від зілля Гермеса      0
  temp_atk_buff   real     +ATK від зілля Геракла      0
  buff_duration   real     Кадри до кінця бафу         0
  current_weapon  int      Поточна зброя (-1=немає)   -1

  ── ФІЗИКА ──────────────────────────────────────────────────────
  hspd      — горизонтальна швидкість (knockback від ударів)
  vspd      — вертикальна швидкість (гравітація + стрибки)
  grv = 0.5 — прискорення падіння кожен кадр

  Реальна швидкість руху:
    _current_spd = spd * weapon_spd_mult[current_weapon] * 10
                   + (perm_spd_bonus + temp_spd_buff) * 10

  Knockback після удару:
    hspd += knockback_x   (призначається ворогом/босом)
    hspd *= 0.75          (загасає кожен кадр)

  ── КЕРУВАННЯ ────────────────────────────────────────────────────
  A / D / ←→    — рух ліво/право
  W / ↑         — стрибок (тільки на землі)
  S / ↓         — проскочити вниз крізь obj_thin_platform
  E             — кинути зброю (Тризубець/Спис) або випад (Меч)
  F             — повернути зброю / взаємодіяти з предметом
  1–6           — пряме переключення зброї
  Колесо миші   — переключення між розблокованими
  Tab           — колесо вибору зброї
  Z             — зілля здоров'я (+5 HP)
  X             — зілля Геракла (+5 ATK на 60с)
  C             — зілля Гермеса (+1 SPD на 60с)
  Q             — закрити діалог
  ESC           — пауза

  ── СТАНИ ГРАВЦЯ ─────────────────────────────────────────────────
  is_damage = true    — гравець щойно отримав удар (невразливість)
  is_frozen = true    — заморожений Тінню Аїда (не може рухатись)
  alarm[0]            — скидає is_damage через 1 секунду

  ── ЯК ЗМІНИТИ БАЗОВІ ХАРАКТЕРИСТИКИ ────────────────────────────
  // Більше HP від початку:
  hp = 7; max_hp = 7;

  // Швидша гравітація (ігра більш "вагома"):
  grv = 0.8;

  // Вищий стрибок:
  vspd = -16;  // (у блоці стрибка в Step Event)

  // Менший knockback:
  // Замінити 0.75 на 0.85 (загасає повільніше = далі відлітає)
  hspd *= 0.85;

  3. СИСТЕМА ЗБРОЇ
════════════════════════════════════════════════════════════════════════════════

  ── ТАБЛИЦЯ ЗБРОЇ ───────────────────────────────────────────────
  Індекс  Зброя        ATK   SPD×    Спецдія               Від боса
  ──────────────────────────────────────────────────────────────────
  0       Тризубець    1     1.0     Кидок, платформа       Тритон
  1       Спис         2     2.0     Кидок, платформа       Німфа
  2       Кайло        5     0.25    Розбиває брили         Уран
  3       Коса         3     1.0     Скошує хащі            Харон
  4       Весло        4     0.5     Плавання на плоту      Мінотавр
  5       Меч          10    2.0     Випад вперед (E)       Привид

  ── ЯК РОЗБЛОКУЄТЬСЯ ЗБРОЯ ──────────────────────────────────────
  1. Гравець вбиває боса
  2. Бос створює obj_pickup_*_reward на місці своєї смерті
  3. Гравець підходить до нагороди — спрацьовує User Event 0
  4. Викликається _unlock_weapon(index) на obj_player
  5. weapon_unlocked[index] = true, current_weapon = index

  ── ЯК ПЕРЕКЛЮЧИТИ ЗБРОЮ ВРУЧНУ (з коду) ───────────────────────
  // Одразу дати гравцю меч:
  with (obj_player) { _unlock_weapon(5); }

  // Переключити на конкретну зброю без розблокування:
  obj_player.current_weapon = 2;  // Кайло

  ── МЕХАНІКА ТРИЗУБЦЯ / СПИСА ───────────────────────────────────
  E    → зброя летить у бік де дивиться гравець (fly_speed = 12)
  Стіна → зброя застряє і стає obj_trident_platform (платформа)
  F/1  → зброя повертається до гравця
  Ворог → зброя завдає шкоди і відскакує

  Об'єкти: obj_trident, obj_trident_platform
  Змінні:  thrown (bool), stuck (bool), fly_speed (real)

  ── ЯК ЗМІНИТИ ПАРАМЕТРИ ЗБРОЇ ──────────────────────────────────
  // У obj_player Create Event:
  weapon_atk      = [1, 2, 5, 3, 4, 10];    // ATK кожної зброї
  weapon_spd_mult = [1.0, 2.0, 0.25, 1.0, 0.5, 2.0]; // множник SPD

  // Наприклад зробити Кайло швидшим:
  weapon_spd_mult[2] = 0.6;   // Було 0.25, стало 0.6

  // Зробити Меч сильнішим:
  weapon_atk[5] = 15;         // Було 10, стало 15

  4. СИСТЕМА ВОРОГІВ
════════════════════════════════════════════════════════════════════════════════

  ── ЗАГАЛЬНА СТРУКТУРА ВОРОГА ────────────────────────────────────
  Кожен ворог наслідує obj_enemy і має:

  Змінна          Опис
  ──────────────────────────────────────────
  hp              Здоров'я
  atk             Шкода при атаці
  spd             Швидкість руху
  state           Поточний стан (0=патруль, 1=переслідує, 2=атака)
  sight_range     Відстань виявлення гравця
  attack_range    Відстань для атаки
  attack_cd       Поточний кулдаун атаки
  attack_cd_max   Кадрів між атаками

  ── СИСТЕМА СТАНІВ (State Machine) ──────────────────────────────
  state = 0  PATROL   — ворог патрулює або стоїть
  state = 1  CHASE    — переслідує гравця
  state = 2  ATTACK   — атакує в ближньому бою
  state = 3  DEAD     — гине (зазвичай одразу instance_destroy)

  Переходи між станами:
    PATROL → CHASE:   distance_to_object(obj_player) <= sight_range
    CHASE  → ATTACK:  distance_to_object(obj_player) <= attack_range
    CHASE  → PATROL:  distance_to_object(obj_player) > sight_range * 1.5
    ATTACK → CHASE:   distance_to_object(obj_player) > attack_range * 1.5

  ── ЗАВДАННЯ ШКОДИ ГРАВЦЮ ───────────────────────────────────────
  Стандартний патерн (однаковий для всіх ворогів):
    if (distance_to_object(obj_player) <= attack_range
        && attack_cd <= 0
        && !obj_player.is_damage)
    {
        obj_player.hp       -= atk;
        var _d = point_direction(x, y, obj_player.x, obj_player.y);
        obj_player.hspd     = lengthdir_x(СИЛА_KNOCKBACK, _d);
        obj_player.vspd     = -СИЛА_ПІДКИДАННЯ;
        obj_player.is_damage = true;
        obj_player.alarm[0]  = room_speed;
        attack_cd = attack_cd_max;
    }

  ── ТАБЛИЦЯ ВОРОГІВ ─────────────────────────────────────────────
  Ворог             HP   ATK  SPD  Особливість          Зона
  ──────────────────────────────────────────────────────────────────
  Гіганський краб   1    1    2    Патруль              Море
  Гарпія-Чайка      2    2    3    Пікірує              Узбережжя
  Дух Шахтаря       5    3    3.5  Стоїть/кидається     Шахти
  Тінь Аїда         20   20   1.5  Заморожує криком     Аїд
  Грецький Воїн     10   10   4    Патруль              Греція
  Дух Полеглого     15   15   4    Лук + меч            Поле Битви
  Вогонь Олімпу     5    5    6    Гине після атаки     Олімп

  ── ЯК ЗМІНИТИ СКЛАДНІСТЬ ВОРОГА ────────────────────────────────
  // Зробити Тінь Аїда менш небезпечною:
  // У obj_hades_shadow Create Event:
  hp           = 10;          // Менше HP (було 20)
  atk          = 10;          // Менша шкода (було 20)
  scream_range = 100;         // Менший радіус крику (було 180)
  scream_cd_max = room_speed * 8; // Рідше кричить (було 4 сек)

  ── ЗАМОРОЗКА (унікальна механіка Тіні Аїда) ────────────────────
  Коли ворог кричить → встановлює на гравці:
    obj_player.is_frozen    = true;
    obj_player.freeze_timer = room_speed;   // 1 секунда
  У Step гравця:
    if (is_frozen) { freeze_timer--; блокуємо рух; }

  5. СИСТЕМА БОСІВ
════════════════════════════════════════════════════════════════════════════════

  ── БАТЬКІВСЬКИЙ ОБ'ЄКТ OBJ_BOSS ───────────────────────────────
  Містить тільки Draw GUI Event зі смугою HP.
  Всі змінні встановлюються у нащадках.

  Обов'язкові змінні у кожному бос-об'єкті:
    boss_name_key  — ключ для lc() ("boss_triton" тощо)
    max_hp         — максимальне HP
    hp             — поточне HP
    phase          — поточна фаза (1 або 2)
    is_dead        — захист від подвійної смерті

  ── СМУГА HP БОСА ───────────────────────────────────────────────
  Малюється автоматично через Draw GUI obj_boss.
  Колір: червоний → жовтий залежно від % HP.
  При hp <= max_hp/2 → phase = 2 → з'являється "— ФАЗА 2 —"

  ── ТАБЛИЦЯ БОСІВ ───────────────────────────────────────────────
  Бос             HP    ATK  Унікальна механіка
  ──────────────────────────────────────────────────────────────────
  Тритон          50    1    Кнопки → каміння (10 шкоди)
  Німфа           10    2    Кінь + залп стріл з пропусками
  Уран            40    5    Завали, застрягання (вразливий)
  Харон           100   3    Коса-бумеранг, сталактити
  Мінотавр        60    4    Колони (5 ударів), весло
  Привид Війни    50    4    Духи полеглих, невразливий вгорі
  Мойри           50    —    Хвиля при шкоді, Вогні Олімпу

  ── ЯК ЗАБЛОКУВАТИ ВИХІД ПІСЛЯ СМЕРТІ БОСА ─────────────────────
  // У Step кожного боса (блок смерті):
  if (hp <= 0 && !is_dead)
  {
      is_dead = true;
      // Розблокуємо вихід:
      with (obj_room_exit)
      {
          if (is_locked) is_locked = false;
      }
      // Видаємо нагороду:
      instance_create_layer(x, y, "Instances", obj_pickup_НАЗВА_reward);
      instance_destroy();
  }

  ── ЯК ЗМІНИТИ HP БОСА ──────────────────────────────────────────
  // У Create Event потрібного боса:
  max_hp = 75;   // Змінити тут
  hp     = max_hp;  // hp завжди = max_hp при старті

  6. СИСТЕМА ПРЕДМЕТІВ
════════════════════════════════════════════════════════════════════════════════

  ── ТАБЛИЦЯ ПРЕДМЕТІВ ───────────────────────────────────────────
  Об'єкт                   Ефект                    Тип
  ──────────────────────────────────────────────────────────────────
  obj_pickup_potion_heal   +5 HP (до max_hp)         Одноразовий
  obj_pickup_potion_str    +5 ATK на 60 секунд       Одноразовий
  obj_pickup_potion_spd    +1 SPD на 60 секунд       Одноразовий
  obj_pickup_ambrosia      +1 max_hp назавжди        Перманентний
  obj_pickup_sandal        +1 SPD назавжди (макс 2)  Перманентний

  ── ЯК ЗІЛЛЯ ПОТРАПЛЯЄ ДО ГРАВЦЯ ───────────────────────────────
  1. Гравець підходить до obj_pickup_potion_*
  2. Зілля потрапляє в інвентар (поtion_heal_count++)
  3. Гравець натискає Z/X/C щоб випити
  4. Ефект застосовується, лічильник зменшується

  ── ЯК ЗМІНИТИ ЕФЕКТ ЗІЛЛЯ ──────────────────────────────────────
  // У obj_player Step Event (блок зілля):

  // Зілля здоров'я відновлює 3 HP замість 5:
  hp = min(hp + 3, max_hp);

  // Зілля Геракла діє 30 секунд замість 60:
  buff_duration = room_speed * 30;

  // Зілля Гермеса дає +2 SPD:
  temp_spd_buff = 2;

  ── ЯК РОЗМІСТИТИ ПРЕДМЕТ У КІМНАТІ ────────────────────────────
  1. Перетягни obj_pickup_potion_heal у кімнату
  2. Більше нічого не потрібно — підбирається автоматично

  ── АМБРОЗІЯ (перманентне HP) ────────────────────────────────────
  При підборі: max_hp++, hp++
  Немає ліміту в поточній системі.
  Щоб додати ліміт:
    if (max_hp < 10) { max_hp++; hp++; }  // Максимум 10 HP

  7. СИСТЕМА ІНТЕРФЕЙСУ (UI)
════════════════════════════════════════════════════════════════════════════════

  ОБ'ЄКТ: obj_ui_controller  |  Persistent: TRUE
  Малює все через Draw GUI Event (GUI-простір, незалежний від камери).

  ── LAYOUT ЕКРАНУ ───────────────────────────────────────────────
  ┌────────────────┬───────────────────┬────────────────┐  ↑ 80px
  │ [?] Ви         │  ЗОНА — №         │  Підказки/[?]  │  верхня
  │  говорите      │  [підказка]       │  Ворог говорить│  панель
  └────────────────┴───────────────────┴────────────────┘  ↓
  ♥♥♥♥♥  [амфора] АТК: X (зброя)  ШВ: X    [баф]
  ... ігрове поле ...
  ┌────────────────────────────────────────────────────┐  ↑ 60px
  │ [Z]×N  [X]×N  [C]×N              [зброя] НАЗВА    │  нижня
  └────────────────────────────────────────────────────┘  ↓

  ── ДІАЛОГОВА СИСТЕМА ───────────────────────────────────────────
  Три типи повідомлень:

  1. Діалог гравця (ліво вгорі):
     obj_ui_controller.show_player_dialog("Текст", тривалість_кадри);
     Закривається: Q або автоматично після тривалості.

  2. Діалог NPC/ворога (право вгорі):
     obj_ui_controller.show_npc_dialog("Текст", тривалість_кадри);

  3. Підказка/завдання (центр вгорі):
     obj_ui_controller.show_hint("Текст", тривалість_кадри);

  Де тривалість: room_speed = 1 секунда.
  Приклади:
    show_player_dialog("Хм...", room_speed * 3);     // 3 секунди
    show_npc_dialog("Зупинись!", room_speed * 4);    // 4 секунди
    show_hint("Натисни E", room_speed * 5);          // 5 секунд

  ── ЗОНА КОМЕНТАРІВ (obj_comment_zone) ──────────────────────────

  8. СИСТЕМА ПЕРЕХОДІВ МІЖ КІМНАТАМИ

  ОБ'ЄКТИ: obj_room_exit, obj_room_transition

  ── ЯК ПРАЦЮЄ ПЕРЕХІД ───────────────────────────────────────────
  1. Гравець входить в зону obj_room_exit
  2. З'являється підказка "[F] Перейти → Назва"
  3. Гравець натискає F
  4. Перевіряються умови (is_locked, required_item)
  5. Створюється obj_room_transition (затемнення)
  6. Зберігаються глобальні spawn-дані
  7. room_goto(target_room) — перехід
  8. У Room Start нової кімнати: гравець з'являється біля

  9. Затемнення зникає

  target_room       room     Кімната призначення

  exit_direction    string   "right" / "left" / "up" / "down"

  exit_height       real     Висота зони переходу (пікселі)
  target_room_name  string   Назва для підказки ("[F] Перейти → Х")
  required_item     string   "" / "pickaxe" / "scythe" / "key" / ...
  is_locked         bool     true = заблокований вручну
  lock_reason       string   Текст чому заблокований

  ── SPAWN_TAG — ЯК СПАРЮВАННЯ ПРАЦЮЄ ────────────────────────────
  spawn_tag — це рядок-ідентифікатор пари "вхід-вихід".

  у новій кімнаті шукається вихід з таким самим тегом "X"

  Приклад: "mines4_to_hades1"

    Parent → obj_enemy   ← ОБОВ'ЯЗКОВО

  КРОК 2 — Create Event:

    atk          = 4;
    spd          = 2;
    sight_range  = 180;
    attack_range = 50;
    attack_cd    = 0;
    attack_cd_max = 90;
    state        = 0;
    // Особлива механіка: отрута
    poison_duration = room_speed * 5;  // 5 секунд отрути

  КРОК 3 — Step Event (копіюй шаблон):
    // Смерть
    if (hp <= 0) { instance_destroy(); exit; }
    if (!instance_exists(obj_player)) exit;
    if (attack_cd > 0) attack_cd--;
    var _dist = distance_to_object(obj_player);

    switch (state)
    {
        case 0: // Патруль (або стоїть)
            if (_dist <= sight_range) state = 1;
        break;
        case 1: // Переслідує (летить — без гравітації)
            var _dir = point_direction(x, y, obj_player.x, obj_player.y);
            x += lengthdir_x(spd, _dir);
            y += lengthdir_y(spd, _dir);
            if (_dist <= attack_range && attack_cd <= 0
                && !obj_player.is_damage)
            {
                obj_player.hp       -= atk;
                // Отрута — зменшує HP кожну секунду
                obj_player.is_poisoned   = true;
                obj_player.poison_timer  = poison_duration;
                obj_player.is_damage     = true;
                obj_player.alarm[0]      = room_speed;
                attack_cd = attack_cd_max;
            }
            if (_dist > sight_range * 1.5) state = 0;
        break;
    }

  КРОК 4 — Додай отруту у obj_player Step Event:
    // Після блоку бафів:
    if (variable_instance_exists(id, "is_poisoned") && is_poisoned)

        if (poison_timer <= 0) is_poisoned = false;

  15. ЧАСТІ ПОМИЛКИ ТА РІШЕННЯ
════════════════════════════════════════════════════════════════════════════════

  ── ПОМИЛКА: Variable not set before reading ────────────────────
  Причина: змінна не ініціалізована у Create Event
  Рішення: перевір Create Event об'єкта — чи є там потрібна змінна

  ── ПОМИЛКА: rm_NAME not set before reading ──────────────────────
  Причина: кімната з такою назвою не існує у проекті
  Рішення: створи кімнату або перевір правильність назви

  ── ПОМИЛКА: Серця відображаються навпаки ────────────────────────
  Причина: hp і max_hp переплутані або hp = 0 при старті
  Рішення: переконайся що в Create Event: hp = 5; max_hp = 5;
           Перевір чи немає десь hp = 0 після Create

  ── ПОМИЛКА: Гравець не з'являється після переходу ───────────────
  Причина: spawn_tag не збігаються або obj_player не persistent
  Рішення: перевір що spawn_tag однаковий в обох кімнатах
           obj_player → Persistent = TRUE в редакторі

  ── ПОМИЛКА: lc() повертає [ключ] замість тексту ─────────────────
  Причина: ключ не додано у словник або obj_lang_manager не існує
  Рішення: додай рядок у Create Event obj_lang_manager
           Перевір що obj_lang_manager є у стартовій кімнаті

  ── ПОМИЛКА: Зброя не переключається клавішами 1-6 ───────────────
  Причина: weapon_unlocked[index] = false
  Рішення: для тестування: weapon_unlocked = [true,true,true,true,true,true];

  ── ПОМИЛКА: Бос не вмирає від тризубця ──────────────────────────
  Причина: obj_boss не встановлений як Parent для боса
  Рішення: у Object Properties кожного боса → Parent = obj_boss
           (або obj_enemy якщо collision з тризубцем на obj_enemy)

  ── ПОМИЛКА: obj_room_transition не зникає після переходу ─────────
  Причина: obj_room_transition не Persistent
  Рішення: obj_room_transition → Persistent = TRUE в редакторі

  ── ПОМИЛКА: ds_map_destroy crash при виході ─────────────────────
  Причина: Cleanup Event не спрацьовує або карта вже знищена
  Рішення: додати перевірку:
           if (ds_exists(global.lang_db, ds_type_map))
               ds_map_destroy(global.lang_db);

  ── ПОРАДИ ───────────────────────────────────────────────────────
