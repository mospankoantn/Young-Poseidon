LAYOUT (на основі малюнку + Game.png + Main_Menu.png):

  ┌──────────────────┬────────────────────┬──────────────────┐
  │ [?] Ви говорите  │   ЗОНАназва – №    │ Підказки/Ворог   │  ← рядок 1
  │ ┌──────────────┐ │ ┌────────────────┐ │ ┌──────────────┐ │
  │ │ текст гравця │ │ │ Підказка/завд. │ │ │ текст NPC    │ │  ← рядок 2
  │ └──────────────┘ │ └────────────────┘ │ └──────────────┘ │
  │ [E]              │                    │ [F]              │  ← рядок 3
  ├──────────────────┴────────────────────┴──────────────────┤
  │ ♥♥♥♥♥   [Амфора] ATK: 1 (Тризубець)  ШВ: 1             │  ← рядок 4
  └──────────────────────────────────────────────────────────┘
  ...гра...
  ┌──────────────────────────────────────────────────────────┐
  │ [Амфора]×N [Геракл]×N [Гермес]×N          [Зброя] НАЗВА │  ← рядок 5 (дно)
  └──────────────────────────────────────────────────────────┘

ЗМІСТ:
  1.  obj_ui_controller — Create Event   (глобальні змінні UI)
  2.  obj_ui_controller — Draw GUI Event (повний перемальований UI)
  3.  obj_dialog_manager — система діалогів гравця та NPC
  4.  obj_hud_hint       — центральна підказка/завдання
  5.  obj_main_menu      — оновлене головне меню (відповідно до Main_Menu.png)


═══════════════════════════════════════════════════════════════════
  1. OBJ_UI_CONTROLLER — Create Event
═══════════════════════════════════════════════════════════════════

─────────────────────────────────────────
/// @description obj_ui_controller — Create Event
─────────────────────────────────────────
// ════ НАЗВА ЗОНИ (рядок вгорі по центру) ════
// Встановлюється при зміні кімнати
zone_name   = "Море";      // Назва зони
zone_room   = 1;           // Номер кімнати в зоні

// ════ ПІДКАЗКА / ЗАВДАННЯ (центр вгорі) ════
hint_text   = "";          // Поточна підказка
hint_timer  = 0;           // Скільки кадрів ще показувати
hint_alpha  = 0;           // Прозорість (плавна поява/зникнення)

// ════ ДІАЛОГ ГРАВЦЯ (ліво вгорі, E) ════
player_dialog_text  = "";
player_dialog_timer = 0;
player_dialog_alpha = 0;

// ════ ДІАЛОГ NPC/ВОРОГА (право вгорі, F) ════
npc_dialog_text     = "";
npc_dialog_timer    = 0;
npc_dialog_alpha    = 0;

// ════ ФУНКЦІЇ ВИКЛИКУ ДІАЛОГІВ ════
// Виклик з будь-якого об'єкта:
//   obj_ui_controller.show_player_dialog("Тут темно...", 180);
//   obj_ui_controller.show_npc_dialog("Зупинись, смертний!", 240);
//   obj_ui_controller.show_hint("Натисни E щоб кинути зброю", 300);

show_player_dialog = function(_text, _duration)
{
    player_dialog_text  = _text;
    player_dialog_timer = _duration;
    player_dialog_alpha = 0;
    // audio_play_sound(snd_dialog_open, 1, false);
};

show_npc_dialog = function(_text, _duration)
{
    npc_dialog_text     = _text;
    npc_dialog_timer    = _duration;
    npc_dialog_alpha    = 0;
    // audio_play_sound(snd_dialog_npc, 1, false);
};

show_hint = function(_text, _duration)
{
    hint_text   = _text;
    hint_timer  = _duration;
    hint_alpha  = 0;
};

// ════ ПАУЗА ════
game_paused = false;

// ════ РОЗМІРИ GUI ════
gw = display_get_gui_width();
gh = display_get_gui_height();


═══════════════════════════════════════════════════════════════════
  1b. OBJ_UI_CONTROLLER — Step Event
═══════════════════════════════════════════════════════════════════

─────────────────────────────────────────
/// @description obj_ui_controller — Step Event
─────────────────────────────────────────
// ---- ДІАЛОГ ГРАВЦЯ ----
if (player_dialog_timer > 0)
{
    player_dialog_timer--;
    player_dialog_alpha = lerp(player_dialog_alpha, 1, 0.15);
    // Q — закрити діалог гравця (з дизайн-документу)
    if (keyboard_check_pressed(ord("Q")))
        player_dialog_timer = 0;
}
else
    player_dialog_alpha = lerp(player_dialog_alpha, 0, 0.12);

// ---- ДІАЛОГ NPC ----
if (npc_dialog_timer > 0)
{
    npc_dialog_timer--;
    npc_dialog_alpha = lerp(npc_dialog_alpha, 1, 0.15);
    if (keyboard_check_pressed(ord("Q")))
        npc_dialog_timer = 0;
}
else
    npc_dialog_alpha = lerp(npc_dialog_alpha, 0, 0.12);

// ---- ПІДКАЗКА/ЗАВДАННЯ ----
if (hint_timer > 0)
{
    hint_timer--;
    hint_alpha = lerp(hint_alpha, 1, 0.15);
}
else
    hint_alpha = lerp(hint_alpha, 0, 0.1);

// ---- ESC = ПАУЗА ----
if (keyboard_check_pressed(vk_escape))
{
    game_paused = !game_paused;
    // audio_play_sound(snd_pause, 1, false);
}

// ---- ОНОВЛЕННЯ РОЗМІРІВ (якщо вікно змінилося) ----
gw = display_get_gui_width();
gh = display_get_gui_height();


═══════════════════════════════════════════════════════════════════
  2. OBJ_UI_CONTROLLER — Draw GUI Event (ПОВНИЙ НОВИЙ UI)
═══════════════════════════════════════════════════════════════════

─────────────────────────────────────────
/// @description obj_ui_controller — Draw GUI Event
─────────────────────────────────────────
if (!instance_exists(obj_player)) exit;
var _p = obj_player;

if (font_exists(fnt_main)) draw_set_font(fnt_main);

var _pad = 10;   // Відступ від країв


// ════════════════════════════════════════════════════════════════
// РЯДОК 1–3: ВЕРХНЯ ПАНЕЛЬ (3 блоки: ліво, центр, право)
// ════════════════════════════════════════════════════════════════
var _top_h   = 80;    // Висота верхньої панелі
var _blk_w   = 280;   // Ширина лівого і правого блоків

// ── ЗАГАЛЬНЕ ТЛО ВЕРХНЬОЇ ПАНЕЛІ ──
draw_set_alpha(0.82);
draw_set_color(make_color_rgb(8, 15, 30));
draw_rectangle(0, 0, gw, _top_h, false);
draw_set_alpha(0.4);
draw_set_color(make_color_rgb(60, 100, 180));
draw_rectangle(0, _top_h - 2, gw, _top_h, false);
draw_set_alpha(1);


// ════════════════════════════════════════
// БЛОК ЛІВИЙ: "Ви говорите" (діалог гравця, E)
// ════════════════════════════════════════
var _lx  = _pad;
var _ly  = _pad / 2;
var _lbw = _blk_w;

// Іконка "?" зліва
draw_set_alpha(1);
draw_set_color(make_color_rgb(80, 140, 220));
draw_rectangle(_lx, _ly + 2, _lx + 22, _ly + 22, false);
draw_set_color(c_white);
draw_set_halign(fa_center);
draw_set_valign(fa_middle);
draw_text(_lx + 11, _ly + 12, "?");

// Заголовок "Ви говорите"
draw_set_color(make_color_rgb(140, 180, 240));
draw_set_halign(fa_left);
draw_set_valign(fa_top);
draw_text(_lx + 28, _ly + 3, "Ви говорите:");

// Рамка діалогу
var _dialog_box_y  = _ly + 22;
var _dialog_box_h  = _top_h - _dialog_box_y - 14;

draw_set_alpha(0.65);
draw_set_color(make_color_rgb(15, 30, 55));
draw_rectangle(_lx, _dialog_box_y, _lx + _lbw, _dialog_box_y + _dialog_box_h, false);
draw_set_alpha(0.5);
draw_set_color(make_color_rgb(60, 100, 180));
draw_rectangle(_lx, _dialog_box_y, _lx + _lbw, _dialog_box_y + _dialog_box_h, true);
draw_set_alpha(1);

// Текст гравця
if (player_dialog_alpha > 0.05)
{
    draw_set_alpha(player_dialog_alpha);
    draw_set_color(c_white);
    draw_set_halign(fa_left);
    draw_set_valign(fa_top);
    draw_text_ext(_lx + 6, _dialog_box_y + 4,
        player_dialog_text, _dialog_box_h - 6, _lbw - 12);
    draw_set_alpha(1);
}
else
{
    // Підказка "натисни E"
    draw_set_alpha(0.35);
    draw_set_color(make_color_rgb(100, 130, 180));
    draw_set_halign(fa_center);
    draw_text(_lx + _lbw / 2, _dialog_box_y + _dialog_box_h / 2 - 6, "[E] — говорити");
    draw_set_alpha(1);
}


// ════════════════════════════════════════
// БЛОК ЦЕНТР: Назва зони + підказка/завдання
// ════════════════════════════════════════
var _cx  = gw / 2;
var _czw = gw - _blk_w * 2 - _pad * 4;

// Назва зони: "МОРЕ – 1"
draw_set_halign(fa_center);
draw_set_valign(fa_top);
draw_set_color(make_color_rgb(180, 210, 255));
draw_text(_cx, _ly + 3,
    string_upper(zone_name) + " — " + string(zone_room));

// Підказка/завдання (під назвою зони)
if (hint_alpha > 0.05)
{
    draw_set_alpha(hint_alpha);

    // Тло підказки
    var _htw = _czw - 20;
    var _htx = _cx - _htw / 2;
    var _hty = _ly + 22;
    var _hth = _top_h - _hty - 14;

    draw_set_color(make_color_rgb(20, 40, 70));
    draw_rectangle(_htx, _hty, _htx + _htw, _hty + _hth, false);
    draw_set_color(make_color_rgb(100, 160, 255));
    draw_rectangle(_htx, _hty, _htx + _htw, _hty + _hth, true);

    draw_set_color(c_white);
    draw_set_halign(fa_center);
    draw_text_ext(_cx, _hty + 4, hint_text, _hth - 6, _htw - 12);
    draw_set_alpha(1);
}
else
{
    // Порожній прямокутник-заглушка
    draw_set_alpha(0.2);
    var _htw = _czw - 20;
    var _htx = _cx - _htw / 2;
    var _hty = _ly + 22;
    var _hth = _top_h - _hty - 14;
    draw_set_color(make_color_rgb(30, 50, 90));
    draw_rectangle(_htx, _hty, _htx + _htw, _hty + _hth, false);
    draw_set_alpha(1);
}


// ════════════════════════════════════════
// БЛОК ПРАВИЙ: "Підказки / Ворог говорить" (F)
// ════════════════════════════════════════
var _rx  = gw - _pad - _blk_w;
var _ry  = _ly;

// Іконка "?" справа
draw_set_alpha(1);
draw_set_color(make_color_rgb(220, 100, 80));
draw_rectangle(_rx + _blk_w - 22, _ry + 2, _rx + _blk_w, _ry + 22, false);
draw_set_color(c_white);
draw_set_halign(fa_center);
draw_set_valign(fa_middle);
draw_text(_rx + _blk_w - 11, _ry + 12, "?");

// Заголовок
draw_set_color(make_color_rgb(240, 160, 120));
draw_set_halign(fa_right);
draw_set_valign(fa_top);
draw_text(_rx + _blk_w - 28, _ry + 3, "Підказки / Ворог:");

// Рамка діалогу NPC
draw_set_alpha(0.65);
draw_set_color(make_color_rgb(35, 15, 10));
draw_rectangle(_rx, _dialog_box_y, _rx + _blk_w, _dialog_box_y + _dialog_box_h, false);
draw_set_alpha(0.5);
draw_set_color(make_color_rgb(180, 80, 60));
draw_rectangle(_rx, _dialog_box_y, _rx + _blk_w, _dialog_box_y + _dialog_box_h, true);
draw_set_alpha(1);

if (npc_dialog_alpha > 0.05)
{
    draw_set_alpha(npc_dialog_alpha);
    draw_set_color(make_color_rgb(255, 220, 180));
    draw_set_halign(fa_left);
    draw_set_valign(fa_top);
    draw_text_ext(_rx + 6, _dialog_box_y + 4,
        npc_dialog_text, _dialog_box_h - 6, _blk_w - 12);
    draw_set_alpha(1);
}
else
{
    draw_set_alpha(0.35);
    draw_set_color(make_color_rgb(160, 100, 80));
    draw_set_halign(fa_center);
    draw_text(_rx + _blk_w / 2, _dialog_box_y + _dialog_box_h / 2 - 6,
        "[F] — взаємодіяти");
    draw_set_alpha(1);
}


// ════════════════════════════════════════════════════════════════
// РЯДОК 4: HP + АМФОРА + ATK / ШВ (під верхньою панеллю, зліва)
// ════════════════════════════════════════════════════════════════
var _hp_y = _top_h + 6;

// Серця (HP)
var _heart_size = 22;
var _heart_gap  = 3;
for (var _hi = 0; _hi < _p.max_hp; _hi++)
{
    var _hx = _pad + _hi * (_heart_size + _heart_gap);
    var _hy = _hp_y;

    if (_hi < _p.hp)
    {
        // Повне серце
        draw_set_color(make_color_rgb(220, 50, 50));
        // Малюємо серце як два кола + трикутник (без спрайту)
        if (sprite_exists(spr_heart_full))
            draw_sprite_ext(spr_heart_full, 0, _hx + _heart_size/2, _hy + _heart_size/2,
                1, 1, 0, c_white, 1);
        else
        {
            draw_set_color(make_color_rgb(220, 50, 50));
            draw_ellipse(_hx + 2, _hy + 2, _hx + 10, _hy + 12, false);
            draw_ellipse(_hx + 10, _hy + 2, _hx + 20, _hy + 12, false);
            draw_triangle(_hx + 2, _hy + 10, _hx + 20, _hy + 10, _hx + 11, _hy + 22, false);
        }
    }
    else
    {
        // Порожнє серце
        if (sprite_exists(spr_heart_empty))
            draw_sprite_ext(spr_heart_empty, 0, _hx + _heart_size/2, _hy + _heart_size/2,
                1, 1, 0, c_white, 1);
        else
        {
            draw_set_color(make_color_rgb(80, 30, 30));
            draw_ellipse(_hx + 2, _hy + 2, _hx + 10, _hy + 12, true);
            draw_ellipse(_hx + 10, _hy + 2, _hx + 20, _hy + 12, true);
            draw_triangle(_hx + 2, _hy + 10, _hx + 20, _hy + 10, _hx + 11, _hy + 22, true);
        }
    }
}

// Амфора + ATK / ШВ (під серцями або праворуч)
var _stats_x = _pad;
var _stats_y = _hp_y + _heart_size + 4;

// Іконка амфори
if (sprite_exists(spr_amphora))
    draw_sprite_ext(spr_amphora, 0, _stats_x + 10, _stats_y + 10, 0.7, 0.7, 0, c_white, 1);

// ATK + Назва зброї
var _atk_total = (_p.current_weapon >= 0)
    ? (_p.weapon_atk[_p.current_weapon] + _p.temp_atk_buff)
    : 0;
var _weapon_label = (_p.current_weapon >= 0)
    ? _p.weapon_names[_p.current_weapon]
    : "Без зброї";

draw_set_halign(fa_left);
draw_set_valign(fa_top);
draw_set_color(make_color_rgb(255, 200, 100));
draw_text(_stats_x + 26, _stats_y,
    "АТК: " + string(_atk_total) + "  (" + _weapon_label + ")");

// ШВ (швидкість)
var _spd_total = _p.spd + _p.perm_spd_bonus + _p.temp_spd_buff;
var _spd_mult  = (_p.current_weapon >= 0) ? _p.weapon_spd_mult[_p.current_weapon] : 1.0;
draw_set_color(make_color_rgb(100, 200, 255));
draw_text(_stats_x + 26, _stats_y + 18,
    "ШВ: " + string(_spd_total * _spd_mult));


// ════════════════════════════════════════════════════════════════
// РЯДОК 4 ПРАВО: Сандалі + Активний баф
// ════════════════════════════════════════════════════════════════

// Сандалі Гермеса (кількість зібраних, макс 2)
var _sandal_x = gw - _pad - 80;
var _sandal_y = _top_h + 6;

for (var _si = 0; _si < 2; _si++)
{
    var _sx    = _sandal_x + _si * 36;
    var _has_s = (_si < _p.perm_spd_bonus);   // є сандаль?

    if (sprite_exists(spr_sandal))
        draw_sprite_ext(spr_sandal, 0, _sx + 14, _sandal_y + 14,
            1, 1, 0, c_white, _has_s ? 1.0 : 0.25);
    else
    {
        draw_set_alpha(_has_s ? 1.0 : 0.25);
        draw_set_color(make_color_rgb(200, 160, 80));
        draw_rectangle(_sx, _sandal_y, _sx + 28, _sandal_y + 22, false);
        draw_set_alpha(1);
    }
}

// Активний баф (якщо є)
if (_p.buff_duration > 0)
{
    var _secs   = ceil(_p.buff_duration / room_speed);
    var _pulse  = 0.65 + 0.35 * sin(current_time / 150000);
    var _bc     = (_p.temp_atk_buff > 0)
                    ? make_color_rgb(255, 100, 60)   // Геракл — вогненний
                    : make_color_rgb(60, 180, 255);  // Гермес — синій

    draw_set_alpha(_pulse);
    draw_set_color(_bc);
    draw_set_halign(fa_right);
    draw_set_valign(fa_top);
    var _buff_str = (_p.temp_atk_buff > 0)
        ? "⚡ Геракл +" + string(_p.temp_atk_buff) + " АТК  [" + string(_secs) + "с]"
        : "💨 Гермес +" + string(_p.temp_spd_buff) + " ШВ  [" + string(_secs) + "с]";
    draw_text(gw - _pad, _top_h + 6, _buff_str);
    draw_set_alpha(1);
    draw_set_halign(fa_left);
}


// ════════════════════════════════════════════════════════════════
// РЯДОК 5: НИЖНЯ ПАНЕЛЬ — Зілля + Зброя
// ════════════════════════════════════════════════════════════════
var _bot_h   = 60;
var _bot_y   = gh - _bot_h;

// Тло нижньої панелі
draw_set_alpha(0.82);
draw_set_color(make_color_rgb(8, 15, 30));
draw_rectangle(0, _bot_y, gw, gh, false);
draw_set_alpha(0.4);
draw_set_color(make_color_rgb(60, 100, 180));
draw_rectangle(0, _bot_y, gw, _bot_y + 2, false);
draw_set_alpha(1);

// ── ЗІЛЛЯ (3 слоти зліва) ──
// Дані: тип, спрайт, лічильник, клавіша, колір
var _potion_data = [
    // [лічильник,   спрайт,              клавіша, колір_фону,             назва]
    [_p.potion_heal_count,     spr_potion_heal,     "Z",
        make_color_rgb(40, 100, 40), "Здоров'я"],
    [_p.potion_strength_count, spr_potion_strength, "X",
        make_color_rgb(100, 40, 20), "Геракл"],
    [_p.potion_speed_count,    spr_potion_speed,    "C",
        make_color_rgb(20, 60, 110), "Гермес"]
];

var _slot_w   = 52;
var _slot_h   = 50;
var _slot_gap = 6;
var _slots_x  = _pad;
var _slots_y  = _bot_y + (_bot_h - _slot_h) / 2;

for (var _pi = 0; _pi < 3; _pi++)
{
    var _count  = _potion_data[_pi][0];
    var _spr    = _potion_data[_pi][1];
    var _key    = _potion_data[_pi][2];
    var _col    = _potion_data[_pi][3];
    var _sx     = _slots_x + _pi * (_slot_w + _slot_gap);
    var _sy     = _slots_y;
    var _empty  = (_count <= 0);

    // Тло слота
    draw_set_alpha(_empty ? 0.3 : 0.85);
    draw_set_color(_col);
    draw_rectangle(_sx, _sy, _sx + _slot_w, _sy + _slot_h, false);

    // Контур
    draw_set_alpha(_empty ? 0.2 : 0.7);
    draw_set_color(_empty ? c_dkgray : c_white);
    draw_rectangle(_sx, _sy, _sx + _slot_w, _sy + _slot_h, true);
    draw_set_alpha(1);

    // Спрайт амфори
    if (sprite_exists(_spr) && !_empty)
        draw_sprite_ext(_spr, 0, _sx + _slot_w/2, _sy + _slot_h/2 - 5,
            1, 1, 0, c_white, 1);
    else if (!_empty)
    {
        // Заглушка — кольоровий прямокутник
        draw_set_color(c_white);
        draw_set_halign(fa_center);
        draw_set_valign(fa_middle);
        draw_text(_sx + _slot_w/2, _sy + _slot_h/2 - 5, "~");
    }

    // Лічильник (знизу справа в слоті)
    draw_set_halign(fa_right);
    draw_set_valign(fa_bottom);
    draw_set_color(_empty ? c_dkgray : c_white);
    draw_text(_sx + _slot_w - 3, _sy + _slot_h - 2, string(_count));

    // Клавіша (знизу зліва в слоті)
    draw_set_halign(fa_left);
    draw_set_color(_empty ? c_dkgray : make_color_rgb(200, 220, 160));
    draw_text(_sx + 3, _sy + _slot_h - 16, "[" + _key + "]");
}
