  2. ФУНКЦІЯ LC() — ГЛОБАЛЬНИЙ ПЕРЕКЛАД
     Додай як Script Asset з назвою: lc
═══════════════════════════════════════════════════════════════════

─────────────────────────────────────────
/// @description Script: lc(key)
/// @param {string} key  Ключ зі словника
/// @returns {string}    Рядок поточної мови
─────────────────────────────────────────
function lc(_key)
{
    // Якщо словник не існує — повертаємо ключ як заглушку
    if (!variable_global_exists("lang_db")
        || !ds_exists(global.lang_db, ds_type_map))
        return _key;

    if (!ds_map_exists(global.lang_db, _key))
    {
        // Ключ не знайдено — попередження і повернення ключа
        show_debug_message("lc(): KEY NOT FOUND — " + _key);
        return "[" + _key + "]";
    }

    var _arr   = global.lang_db[? _key];
    var _index = (global.lang == "UA") ? 0 : 1;

    return _arr[_index];
}

/// @description Script: set_language(lang)
/// @param {string} lang  "UA" або "EN"
─────────────────────────────────────────
function set_language(_lang)
{
    global.lang = _lang;

    // Зберігаємо вибір у файлі
    ini_open("young_poseidon_save.ini");
    ini_write_string("settings", "language", _lang);
    ini_close();

    // Оновлюємо масиви зброї у гравця якщо він існує
    if (instance_exists(obj_player))
        _update_player_weapon_names();

    // Оновлюємо назви кімнат у obj_ui_controller
    if (instance_exists(obj_ui_controller))
        obj_ui_controller._refresh_zone_name();
}


─────────────────────────────────────────
/// @description Script: _update_player_weapon_names()
// Оновлює масив weapon_names у гравця після зміни мови
─────────────────────────────────────────
function _update_player_weapon_names()
{
    if (!instance_exists(obj_player)) exit;
    with (obj_player)
    {
        weapon_names = [
            lc("weapon_trident"),
            lc("weapon_spear"),
            lc("weapon_pickaxe"),
            lc("weapon_scythe"),
            lc("weapon_oar"),
            lc("weapon_sword")
        ];
    }
}

  13. НАЛАШТУВАННЯ obj_lang_manager В РЕДАКТОРІ
═══════════════════════════════════════════════════════════════════

  1. Створи об'єкт obj_lang_manager
     → Persistent = TRUE
     → Розміщуй ОДИН РАЗ у стартовій кімнаті (rm_main_menu)
     → Events: Create, Step (порожній), Cleanup

  2. Створи Script Asset з назвою "lc"
     → Вміст: функція lc(_key) з розділу 2

  3. Створи Script Asset "set_language"
     → Вміст: функція set_language(_lang) з розділу 3

  4. Створи Script Asset "_update_player_weapon_names"
     → Вміст: однойменна функція з розділу 3

  5. У obj_master Create Event переконайся що obj_lang_manager
     вже існує ДО того як створюються інші об'єкти.
     Якщо ні — створи його першим:
     if (!instance_exists(obj_lang_manager))
         instance_create_layer(0, 0, "Instances", obj_lang_manager);

  6. ПОРЯДОК ініціалізації при старті гри:
     obj_lang_manager → lc() доступна → obj_player (weapon_names) → UI
